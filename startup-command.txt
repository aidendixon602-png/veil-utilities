# ================================================================
# Wispbyte / Pterodactyl startup commands
# ================================================================
# Paste ONE of the single-line commands below into your panel:
# Startup tab -> "Startup Command".
#
# main.py already pulls the latest commit from GitHub on every boot (set AUTO_SYNC_GIT=0 in .env to turn that
# off), so these commands only need to install packages when requirements.txt changes and then start the bot.

# ----------------------------------------------------------------
# OPTION 1 (recommended): files directly in /home/container
# Installs packages only when requirements.txt changed, then starts the bot.
# ----------------------------------------------------------------
cd /home/container && if [ -f requirements.txt ] && ! cmp -s requirements.txt .requirements.hash 2>/dev/null; then pip install --prefix .local -r requirements.txt && cp requirements.txt .requirements.hash; fi; python main.py

# ----------------------------------------------------------------
# OPTION 2: files in a subfolder (/home/container/veil-utilities)
# ----------------------------------------------------------------
cd /home/container/veil-utilities && if [ -f requirements.txt ] && ! cmp -s requirements.txt .requirements.hash 2>/dev/null; then pip install --prefix .local -r requirements.txt && cp requirements.txt .requirements.hash; fi; python main.py

# ----------------------------------------------------------------
# OPTION 3: fastest boot, never installs packages
# Only use this once everything is installed; new packages won't be picked up.
# ----------------------------------------------------------------
cd /home/container && python main.py
